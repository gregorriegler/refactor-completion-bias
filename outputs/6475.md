```python
import random
from dataclasses import dataclass, field
from enum import Enum, auto
from typing import Any


class Phase(Enum):
    SETUP = auto()
    ACTIVE = auto()
    VICTORY = auto()
    DEFEAT = auto()
    FLED = auto()


class Side(Enum):
    PLAYER = "p"
    ENEMY = "e"


@dataclass
class Effect:
    kind: str
    power: int
    duration: int


@dataclass
class Combatant:
    name: str
    max_hp: int
    hp: int = 0
    max_mp: int = 0
    mp: int = 0
    dex: int = 0
    atk: int = 0
    ac: int = 10
    dmg_die: int = 6
    dmg_bonus: int = 0
    dmg_type: str = "physical"
    resist: dict[str, float] = field(default_factory=dict)
    drops: list[Any] = field(default_factory=list)
    inventory: dict[str, int] = field(default_factory=dict)
    alive: bool = True
    status: list[Effect] = field(default_factory=list)
    init: int = 0

    def reset_for_combat(self) -> None:
        self.hp = self.max_hp
        self.mp = self.max_mp
        self.alive = True
        self.status = []

    def apply_damage(self, amount: int) -> None:
        self.hp -= amount
        if self.hp <= 0:
            self.alive = False

    def apply_heal(self, amount: int) -> None:
        self.hp = min(self.max_hp, self.hp + amount)

    def tick_effects(self, log: list[str]) -> bool:
        """Apply status effects; returns True if actor is dead after ticking."""
        expired = []
        for eff in list(self.status):
            if eff.kind == "poison":
                self.apply_damage(eff.power)
                log.append(f"{self.name} takes {eff.power} poison")
                eff.duration -= 1
                if eff.duration <= 0:
                    expired.append(eff)
                if not self.alive:
                    log.append(f"{self.name} dies of poison")
                    self.status = [e for e in self.status if e not in expired]
                    return True

            elif eff.kind == "stun":
                eff.duration -= 1
                if eff.duration <= 0:
                    expired.append(eff)
                log.append(f"{self.name} is stunned")
                self.status = [e for e in self.status if e not in expired]
                return False  # caller checks stun separately via _is_stunned

            elif eff.kind == "regen":
                heal = min(eff.power, self.max_hp - self.hp)
                self.apply_heal(heal)
                log.append(f"{self.name} regens {heal}")
                eff.duration -= 1
                if eff.duration <= 0:
                    expired.append(eff)

        self.status = [e for e in self.status if e not in expired]
        return False

    def is_stunned_after_tick(self) -> bool:
        """Separate stun check so tick_effects can still remove the effect."""
        return any(e.kind == "stun" for e in self.status)


class Combat:
    def __init__(
        self,
        players: list[Combatant],
        enemies: list[Combatant],
        terrain: dict[str, Any],
        seed: int | None = None,
    ) -> None:
        self.players = players
        self.enemies = enemies
        self.terrain = terrain
        self.turn = 0
        self.round = 1
        self.log: list[str] = []
        self.loot: list[Any] = []
        self.phase = Phase.SETUP
        self.rng = random.Random(seed)
        self.initiative: list[tuple[Side, int]] = []

    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------

    def start(self) -> None:
        if self.phase != Phase.SETUP:
            raise RuntimeError("already started")
        for combatant in self.players + self.enemies:
            combatant.reset_for_combat()
            combatant.init = self.rng.randint(1, 20) + combatant.dex

        self.initiative = (
            [(Side.PLAYER, i) for i in range(len(self.players))]
            + [(Side.ENEMY, i) for i in range(len(self.enemies))]
        )
        self.initiative.sort(key=lambda ref: -self._combatant(*ref).init)
        self.phase = Phase.ACTIVE
        self.log.append(f"combat start: {len(self.players)}v{len(self.enemies)}")

    # ------------------------------------------------------------------
    # Core turn loop
    # ------------------------------------------------------------------

    def current_actor(self) -> tuple[Side, int] | None:
        if self.phase != Phase.ACTIVE:
            return None
        return self.initiative[self.turn % len(self.initiative)]

    def take_turn(self, action: dict[str, Any]) -> None:
        if self.phase != Phase.ACTIVE:
            raise RuntimeError("not active")

        side, idx = self.current_actor()
        actor = self._combatant(side, idx)

        if not actor.alive:
            self._advance_turn()
            return

        dead_from_effects = actor.tick_effects(self.log)
        if dead_from_effects:
            self._advance_turn()
            self._check_end()
            return

        # Stun: actor was stunned but tick already decremented duration.
        # Re-check after tick — if still stunned the actor loses the turn.
        # (tick_effects returns False for stun but leaves the reduced-duration
        #  effect in place if duration > 0.)
        if any(e.kind == "stun" for e in actor.status) or self._was_just_stunned(actor):
            self._advance_turn()
            self._maybe_end_round()
            return

        kind = action.get("kind")
        if kind == "attack":
            self._do_attack(actor, side, action)
        elif kind == "cast":
            self._do_cast(actor, side, action)
        elif kind == "item":
            self._do_item(actor, action)
        elif kind == "flee":
            self._do_flee(actor, side)
        else:
            self.log.append(f"unknown action {kind}")

        self._advance_turn()
        self._check_end()
        self._maybe_end_round()

    # ------------------------------------------------------------------
    # Action handlers
    # ------------------------------------------------------------------

    def _do_attack(self, actor: Combatant, side: Side, action: dict) -> None:
        target_side = Side.ENEMY if side == Side.PLAYER else Side.PLAYER
        target_list = self._list(target_side)
        tidx = action.get("target", 0)

        if tidx < 0 or tidx >= len(target_list) or not target_list[tidx].alive:
            self.log.append(f"{actor.name} attacks invalid target")
            return

        target = target_list[tidx]
        hit_roll = self.rng.randint(1, 20) + actor.atk
        ac = target.ac
        if self.terrain.get("cover") and target_side == Side.PLAYER:
            ac += 2
        if self.terrain.get("high_ground") == side.value:
            hit_roll += 2

        if hit_roll < ac:
            self.log.append(f"{actor.name} misses {target.name}")
            return

        dmg = self.rng.randint(1, actor.dmg_die) + actor.dmg_bonus
        raw_roll = hit_roll - actor.atk
        if raw_roll == 20:
            dmg *= 2
            self.log.append("CRIT!")

        if actor.dmg_type in target.resist:
            dmg = int(dmg * (1 - target.resist[actor.dmg_type]))

        target.apply_damage(dmg)
        self.log.append(f"{actor.name} hits {target.name} for {dmg}")
        if not target.alive:
            self.log.append(f"{target.name} falls")
            if target_side == Side.ENEMY:
                self.loot.extend(target.drops)

    def _do_cast(self, actor: Combatant, side: Side, action: dict) -> None:
        spell = action.get("spell")
        cost = action.get("cost", 0)
        if actor.mp < cost:
            self.log.append(f"{actor.name} fizzles (no mp)")
            return

        actor.mp -= cost
        enemies_of_actor = self._list(Side.ENEMY if side == Side.PLAYER else Side.PLAYER)
        allies_of_actor = self._list(side)

        if spell == "fireball":
            self._cast_fireball(actor, enemies_of_actor, side)
        elif spell == "heal":
            self._cast_heal(actor, allies_of_actor, action)
        elif spell == "poison_cloud":
            self._cast_poison_cloud(enemies_of_actor, side)
        else:
            self.log.append(f"unknown spell {spell}")

    def _cast_fireball(
        self, actor: Combatant, targets: list[Combatant], caster_side: Side
    ) -> None:
        for t in targets:
            if not t.alive:
                continue
            dmg = self.rng.randint(10, 20)
            if "fire" in t.resist:
                dmg = int(dmg * (1 - t.resist["fire"]))
            t.apply_damage(dmg)
            self.log.append(f"fireball hits {t.name} for {dmg}")
            if not t.alive and caster_side == Side.PLAYER:
                self.loot.extend(t.drops)

    def _cast_heal(
        self, actor: Combatant, allies: list[Combatant], action: dict
    ) -> None:
        tidx = action.get("target", allies.index(actor) if actor in allies else 0)
        tgt = allies[tidx]
        heal = self.rng.randint(8, 16)
        tgt.apply_heal(heal)
        self.log.append(f"{actor.name} heals {tgt.name} for {heal}")

    def _cast_poison_cloud(self, targets: list[Combatant], caster_side: Side) -> None:
        for t in targets:
            if t.alive:
                t.status.append(Effect(kind="poison", power=3, duration=3))
                self.log.append(f"{t.name} is poisoned")

    def _do_item(self, actor: Combatant, action: dict) -> None:
        item = action.get("item")
        if item not in actor.inventory:
            self.log.append(f"{actor.name} has no {item}")
            return

        actor.inventory[item] -= 1
        if actor.inventory[item] <= 0:
            del actor.inventory[item]

        if item == "potion":
            heal = 15
            actor.apply_heal(heal)
            self.log.append(f"{actor.name} drinks potion (+{heal})")
        elif item == "antidote":
            actor.status = [s for s in actor.status if s.kind != "poison"]
            self.log.append(f"{actor.name} uses antidote")
        elif item == "smoke_bomb":
            self.terrain["cover"] = True
            self.log.append("smoke fills the area")
        else:
            self.log.append(f"unknown item {item}")

    def _do_flee(self, actor: Combatant, side: Side) -> None:
        if side != Side.PLAYER:
            self.log.append("enemies don't flee")
            return
        if self.rng.randint(1, 20) >= 12:
            self.phase = Phase.FLED
            self.log.append(f"{actor.name} flees successfully")
        else:
            self.log.append(f"{actor.name} fails to flee")

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------

    def _combatant(self, side: Side, idx: int) -> Combatant:
        return self.players[idx] if side == Side.PLAYER else self.enemies[idx]

    def _list(self, side: Side) -> list[Combatant]:
        return self.players if side == Side.PLAYER else self.enemies

    def _advance_turn(self) -> None:
        self.turn += 1

    def _maybe_end_round(self) -> None:
        if self.turn > 0 and self.turn % len(self.initiative) == 0:
            self.round += 1
            self.log.append(f"round {self.round}")

    def _check_end(self) -> None:
        if not any(p.alive for p in self.players):
            self.phase = Phase.DEFEAT
            self.log.append("party defeated")
        elif not any(e.alive for e in self.enemies):
            self.phase = Phase.VICTORY
            self.log.append(f"victory! loot: {self.loot}")

    @staticmethod
    def _was_just_stunned(actor: Combatant) -> bool:
        """Placeholder: tick_effects already handles stun inline; kept for symmetry."""
        return False
```

- Replaced raw `dict` combatant data with a typed `Combatant` dataclass; all mutable defaults use `field(default_factory=...)`.
- Introduced `Phase` and `Side` enums to eliminate magic strings (`"setup"`, `"p"`, `"e"`, etc.) and make state transitions explicit.
- Extracted an `Effect` dataclass to replace untyped status-effect dicts.
- Moved per-combatant logic (`reset_for_combat`, `apply_damage`, `apply_heal`, `tick_effects`) onto `Combatant`, reducing the size and cognitive load of `take_turn`.
- Decomposed the monolithic `take_turn` into focused action handlers (`_do_attack`, `_do_cast`, `_do_item`, `_do_flee`) and spell sub-handlers (`_cast_fireball`, `_cast_heal`, `_cast_poison_cloud`).
- Added `_combatant` and `_list` helpers to eliminate repeated ternary lookups on `side`.
- Replaced all string concatenation in log messages with f-strings.
- Added type annotations throughout (`dict[str, Any]`, `list[Combatant]`, return types).
