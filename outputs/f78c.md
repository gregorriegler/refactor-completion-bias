```python
import random
from dataclasses import dataclass, field
from enum import Enum, auto
from typing import Optional


class Phase(Enum):
    SETUP = auto()
    ACTIVE = auto()
    VICTORY = auto()
    DEFEAT = auto()
    FLED = auto()


class Side(Enum):
    PLAYER = "p"
    ENEMY = "e"


@dataclass
class StatusEffect:
    kind: str
    power: int
    duration: int


@dataclass
class Combatant:
    name: str
    max_hp: int
    hp: int = 0
    max_mp: int = 0
    mp: int = 0
    dex: int = 0
    atk: int = 0
    ac: int = 10
    dmg_die: int = 6
    dmg_bonus: int = 0
    dmg_type: str = "physical"
    resist: dict = field(default_factory=dict)
    drops: list = field(default_factory=list)
    inventory: dict = field(default_factory=dict)
    alive: bool = True
    status: list[StatusEffect] = field(default_factory=list)
    init: int = 0

    def reset_for_combat(self, rng: random.Random) -> None:
        self.hp = self.max_hp
        self.mp = self.max_mp
        self.alive = True
        self.status = []
        self.init = rng.randint(1, 20) + self.dex

    def apply_damage(self, amount: int, log: list[str]) -> bool:
        """Apply damage, return True if combatant died."""
        self.hp -= amount
        if self.hp <= 0:
            self.alive = False
            return True
        return False

    def apply_resist(self, dmg: int, dtype: str) -> int:
        if dtype in self.resist:
            return int(dmg * (1 - self.resist[dtype]))
        return dmg

    def tick_status(self, log: list[str]) -> Optional[str]:
        """
        Process status effects for the start of this combatant's turn.
        Returns 'dead', 'stunned', or None.
        """
        expired = []
        result = None

        for eff in list(self.status):
            if eff.kind == "poison":
                self.hp -= eff.power
                log.append(f"{self.name} takes {eff.power} poison")
                eff.duration -= 1
                if eff.duration <= 0:
                    expired.append(eff)
                if self.hp <= 0:
                    self.alive = False
                    log.append(f"{self.name} dies of poison")
                    result = "dead"
                    break

            elif eff.kind == "stun":
                eff.duration -= 1
                if eff.duration <= 0:
                    expired.append(eff)
                log.append(f"{self.name} is stunned")
                result = "stunned"
                break

            elif eff.kind == "regen":
                heal = min(eff.power, self.max_hp - self.hp)
                self.hp += heal
                log.append(f"{self.name} regens {heal}")
                eff.duration -= 1
                if eff.duration <= 0:
                    expired.append(eff)

        for eff in expired:
            if eff in self.status:
                self.status.remove(eff)

        return result


class Combat:
    def __init__(
        self,
        players: list[Combatant],
        enemies: list[Combatant],
        terrain: dict,
        seed: Optional[int] = None,
    ):
        self.players = players
        self.enemies = enemies
        self.terrain = terrain
        self.turn = 0
        self.round = 1
        self.log: list[str] = []
        self.loot: list = []
        self.phase = Phase.SETUP
        self.rng = random.Random(seed)
        self.initiative: list[tuple[Side, int]] = []

    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------

    def start(self) -> None:
        if self.phase != Phase.SETUP:
            raise RuntimeError("already started")

        for p in self.players:
            p.reset_for_combat(self.rng)
        for e in self.enemies:
            e.reset_for_combat(self.rng)

        self.initiative = (
            [(Side.PLAYER, i) for i in range(len(self.players))]
            + [(Side.ENEMY, i) for i in range(len(self.enemies))]
        )
        self.initiative.sort(key=lambda ref: -self._get_combatant(*ref).init)

        self.phase = Phase.ACTIVE
        self.log.append(f"combat start: {len(self.players)}v{len(self.enemies)}")

    # ------------------------------------------------------------------
    # Turn management
    # ------------------------------------------------------------------

    def current_actor(self) -> Optional[tuple[Side, int]]:
        if self.phase != Phase.ACTIVE:
            return None
        return self.initiative[self.turn % len(self.initiative)]

    def take_turn(self, action: dict) -> None:
        if self.phase != Phase.ACTIVE:
            raise RuntimeError("not active")

        side, idx = self.current_actor()
        actor = self._get_combatant(side, idx)

        if not actor.alive:
            self._advance_turn()
            return

        status_result = actor.tick_status(self.log)
        if status_result == "dead":
            self._advance_turn()
            self._check_end()
            return
        if status_result == "stunned":
            self._advance_turn()
            return

        handler = {
            "attack": self._handle_attack,
            "cast": self._handle_cast,
            "item": self._handle_item,
            "flee": self._handle_flee,
        }.get(action.get("kind"))

        if handler is None:
            self.log.append(f"unknown action {action.get('kind')}")
        else:
            handler(actor, side, action)

        if self.phase == Phase.FLED:
            return

        self._advance_turn()
        self._check_end()

    # ------------------------------------------------------------------
    # Action handlers
    # ------------------------------------------------------------------

    def _handle_attack(self, actor: Combatant, side: Side, action: dict) -> None:
        target_side = Side.ENEMY if side == Side.PLAYER else Side.PLAYER
        target_list = self._side_list(target_side)
        tidx = action.get("target", 0)

        if tidx < 0 or tidx >= len(target_list) or not target_list[tidx].alive:
            self.log.append(f"{actor.name} attacks invalid target")
            return

        target = target_list[tidx]
        hit_roll = self.rng.randint(1, 20) + actor.atk
        ac = target.ac

        if self.terrain.get("cover") and target_side == Side.PLAYER:
            ac += 2
        if self.terrain.get("high_ground") == side.value:
            hit_roll += 2

        if hit_roll < ac:
            self.log.append(f"{actor.name} misses {target.name}")
            return

        dmg = self.rng.randint(1, actor.dmg_die) + actor.dmg_bonus
        raw_roll = hit_roll - actor.atk
        if raw_roll == 20:
            dmg *= 2
            self.log.append("CRIT!")

        dmg = target.apply_resist(dmg, actor.dmg_type)
        self.log.append(f"{actor.name} hits {target.name} for {dmg}")

        if target.apply_damage(dmg, self.log):
            self.log.append(f"{target.name} falls")
            if target_side == Side.ENEMY:
                self.loot.extend(target.drops)

    def _handle_cast(self, actor: Combatant, side: Side, action: dict) -> None:
        spell = action.get("spell")
        cost = action.get("cost", 0)

        if actor.mp < cost:
            self.log.append(f"{actor.name} fizzles (no mp)")
            return

        actor.mp -= cost
        spell_method = {
            "fireball": self._spell_fireball,
            "heal": self._spell_heal,
            "poison_cloud": self._spell_poison_cloud,
        }.get(spell)

        if spell_method is None:
            self.log.append(f"unknown spell {spell}")
        else:
            spell_method(actor, side, action)

    def _spell_fireball(self, actor: Combatant, side: Side, action: dict) -> None:
        targets = self._opponent_list(side)
        for t in targets:
            if not t.alive:
                continue
            dmg = t.apply_resist(self.rng.randint(10, 20), "fire")
            self.log.append(f"fireball hits {t.name} for {dmg}")
            if t.apply_damage(dmg, self.log):
                if side == Side.PLAYER:
                    self.loot.extend(t.drops)

    def _spell_heal(self, actor: Combatant, side: Side, action: dict) -> None:
        allies = self._side_list(side)
        tidx = action.get("target", self._side_list(side).index(actor))
        tgt = allies[tidx]
        heal = self.rng.randint(8, 16)
        tgt.hp = min(tgt.max_hp, tgt.hp + heal)
        self.log.append(f"{actor.name} heals {tgt.name} for {heal}")

    def _spell_poison_cloud(self, actor: Combatant, side: Side, action: dict) -> None:
        for t in self._opponent_list(side):
            if t.alive:
                t.status.append(StatusEffect(kind="poison", power=3, duration=3))
                self.log.append(f"{t.name} is poisoned")

    def _handle_item(self, actor: Combatant, side: Side, action: dict) -> None:
        item = action.get("item")
        if item not in actor.inventory:
            self.log.append(f"{actor.name} has no {item}")
            return

        actor.inventory[item] -= 1
        if actor.inventory[item] <= 0:
            del actor.inventory[item]

        if item == "potion":
            heal = 15
            actor.hp = min(actor.max_hp, actor.hp + heal)
            self.log.append(f"{actor.name} drinks potion (+{heal})")
        elif item == "antidote":
            actor.status = [s for s in actor.status if s.kind != "poison"]
            self.log.append(f"{actor.name} uses antidote")
        elif item == "smoke_bomb":
            self.terrain["cover"] = True
            self.log.append("smoke fills the area")
        else:
            self.log.append(f"unknown item {item}")

    def _handle_flee(self, actor: Combatant, side: Side, action: dict) -> None:
        if side != Side.PLAYER:
            self.log.append("enemies don't flee")
            return
        if self.rng.randint(1, 20) >= 12:
            self.phase = Phase.FLED
            self.log.append(f"{actor.name} flees successfully")
        else:
            self.log.append(f"{actor.name} fails to flee")

    # ------------------------------------------------------------------
    # Internal helpers
    # ------------------------------------------------------------------

    def _get_combatant(self, side: Side, idx: int) -> Combatant:
        return self.players[idx] if side == Side.PLAYER else self.enemies[idx]

    def _side_list(self, side: Side) -> list[Combatant]:
        return self.players if side == Side.PLAYER else self.enemies

    def _opponent_list(self, side: Side) -> list[Combatant]:
        return self.enemies if side == Side.PLAYER else self.players

    def _advance_turn(self) -> None:
        self.turn += 1
        if self.turn % len(self.initiative) == 0:
            self.round += 1
            self.log.append(f"round {self.round}")

    def _check_end(self) -> None:
        if not any(p.alive for p in self.players):
            self.phase = Phase.DEFEAT
            self.log.append("party defeated")
        elif not any(e.alive for e in self.enemies):
            self.phase = Phase.VICTORY
            self.log.append(f"victory! loot: {self.loot}")
```

- Introduced `Phase` and `Side` enums to replace raw strings like `"setup"`, `"p"`, `"e"`, eliminating silent typo bugs.
- Extracted a `Combatant` dataclass to replace bare dicts; fields are explicit, typed, and carry a `reset_for_combat` method instead of scattering initialization across `start()`.
- Added `StatusEffect` dataclass so effect fields (`kind`, `power`, `duration`) are named and typed rather than accessed by string key.
- Pulled status-tick logic into `Combatant.tick_status`, and damage/resist helpers into `apply_damage`/`apply_resist`, shrinking `take_turn` significantly.
- Decomposed the monolithic `take_turn` into a dispatch table plus dedicated `_handle_attack`, `_handle_cast`, `_handle_item`, `_handle_flee` methods, and each spell into its own `_spell_*` method.
- Collapsed duplicated turn-advance logic into `_advance_turn`, which now owns both the `turn` increment and the round-boundary check; removed the separate `_maybe_end_round`.
- Added `_get_combatant`, `_side_list`, and `_opponent_list` helpers to avoid repeated ternary side-selection throughout the action handlers.
- Replaced all string concatenation in log messages with f-strings.
